236 void reset(T* newPtr = nullptr)
237 {
238 // Disminuir el recuento de referencias del objeto actual
239 if (refCount && --(*refCount) == 0)
240 {
241 delete ptr;
242 delete refCount;
243 }
244
245 // Si newPtr es nullptr, asignar nullptr al puntero y recuento de referencias
246 if (newPtr == nullptr)
247 {
248 ptr = nullptr;
249 refCount = nullptr;
250 }
251 else
252 {
253 // Asignar nuevo objeto y manejar el recuento de referencias
254 ptr = newPtr;
255 refCount = new int(1);
256 }

262 // Intenta convertir el puntero de tipo T a U

264 if (castedPtr) {
